    function tone() {

      if (!audioContext || alarmMuted) return;

      const oscillator = audioContext.createOscillator();

      const gain = audioContext.createGain();

      oscillator.type = "sine";

      oscillator.frequency.value = 880;

      gain.gain.setValueAtTime(.0001, audioContext.currentTime);

      gain.gain.exponentialRampToValueAtTime(.12, audioContext.currentTime + .02);

      gain.gain.exponentialRampToValueAtTime(.0001, audioContext.currentTime + .35);

      oscillator.connect(gain);

      gain.connect(audioContext.destination);

      oscillator.start();

      oscillator.stop(audioContext.currentTime + .38);

    }

    }

 

    function updateDashboard() {

      const scores = records.map(record => Number(record.pain_score)).filter(Number.isFinite);

      const responseTimes = records.map(record => Number(record.nurse_response_time)).filter(time => Number.isFinite(time) && time >= 0);

      document.getElementById("dashboard-total").textContent = scores.length;

      document.getElementById("dashboard-average").textContent = scores.length ? (scores.reduce((sum, score) => sum + score, 0) / scores.length).toFixed(1) : "0.0";

      document.getElementById("dashboard-severe").textContent = scores.filter(score => score >= 7).length;

      document.getElementById("dashboard-response-average").textContent = responseTimes.length ? formatSeconds(responseTimes.reduce((sum, time) => sum + time, 0) / responseTimes.length) : "–";

      document.getElementById("dashboard-response-longest").textContent = responseTimes.length ? formatSeconds(Math.max(...responseTimes)) : "–";

    }

 

    function renderNurse() {

      updateDashboard();

      activeRecord = records.filter(record => Number(record.nurse_response_time) === -1).sort((a, b) => new Date(b.timestamp) - new Date(a.timestamp))[0] || null;

      const activeAlert = document.getElementById("active-alert");

      const emptyAlert = document.getElementById("empty-alert");

      clearInterval(timerInterval);

 

      if (!activeRecord) {

        activeAlert.classList.add("hidden");

        emptyAlert.classList.remove("hidden");

        stopAlarm();

        return;

      }

 

      emptyAlert.classList.add("hidden");

      activeAlert.classList.remove("hidden");

      document.getElementById("nurse-score").textContent = activeRecord.pain_score;

      document.getElementById("nurse-level").textContent = activeRecord.pain_level;

      document.getElementById("nurse-status").textContent = "แจ้งเมื่อ " + new Date(activeRecord.timestamp).toLocaleTimeString("th-TH", { hour:"2-digit", minute:"2-digit" });

      updateTimer();

      timerInterval = setInterval(updateTimer, 1000);

    }

 

    async function submitPain(score, button) {

      if (records.length >= 999) {

        document.getElementById("patient-message").textContent = "ไม่สามารถบันทึกเพิ่มเติมได้ กรุณาแจ้งพยาบาลโดยตรง";

        return;

      }

 

      button.disabled = true;

      document.getElementById("patient-message").textContent = "";

      showWaiting(score);

      startAlarm();

 

      const result = await window.dataSdk.create({

        pain_score: score,

        pain_level: "คะแนน " + score,

        timestamp: new Date().toISOString(),

        nurse_response_time: -1

      });

 

      button.disabled = false;

      if (!result.isOk) {

        stopAlarm();

        showPatient();

        document.getElementById("patient-message").textContent = "ไม่สามารถส่งการแจ้งเตือนได้ กรุณาลองอีกครั้งหรือแจ้งพยาบาลโดยตรง";

      }

    }

 

    async function completeCare() {

      if (!activeRecord) return;

      const button = document.getElementById("care-button");

      button.disabled = true;

      document.getElementById("nurse-status").textContent = "กำลังบันทึกการให้การพยาบาล...";

 

      const responseTime = Math.max(0, Math.round((Date.now() - new Date(activeRecord.timestamp).getTime()) / 1000));

      const result = await window.dataSdk.update({ ...activeRecord, nurse_response_time: responseTime });

 

      button.disabled = false;

      if (!result.isOk) {

        document.getElementById("nurse-status").textContent = "บันทึกไม่สำเร็จ กรุณาลองอีกครั้ง";

      } else {

        stopAlarm();

      }

    }

 

    document.querySelectorAll("[data-score]").forEach(button => {

      button.addEventListener("click", () => submitPain(Number(button.dataset.score), button));

    });

 

    patientTab.addEventListener("click", showPatient);

    nurseTab.addEventListener("click", showNurse);

    document.getElementById("return-button").addEventListener("click", showPatient);

    document.getElementById("mute-button").addEventListener("click", () => {

      alarmMuted = true;

      stopAlarm();

      document.getElementById("nurse-status").textContent = "รับเรื่องแล้ว เสียงแจ้งเตือนหยุดลง กรุณาเข้าดูแลผู้ป่วย";

    });

    document.getElementById("care-button").addEventListener("click", completeCare);

 

    const dataHandler = {

      onDataChanged(data) {

        const previousActiveId = activeRecord && activeRecord.__backendId;

        records = data;

        renderNurse();

        if (activeRecord && activeRecord.__backendId !== previousActiveId && !alarmMuted) startAlarm();

      }

    };

 

    async function initialize() {

      const result = await window.dataSdk.init(dataHandler);

      if (!result.isOk) document.getElementById("patient-message").textContent = "ระบบบันทึกข้อมูลยังไม่พร้อม กรุณาแจ้งพยาบาลโดยตรง";

    }

 

    document.addEventListener("DOMContentLoaded", () => {

      lucide.createIcons();

      initialize();

    });
